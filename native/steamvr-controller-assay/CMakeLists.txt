cmake_minimum_required(VERSION 3.24)
project(csx_controller_atomics LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 acceptance helper only")
endif()
add_library(csx_controller_atomics SHARED atomics.cpp)
target_compile_features(csx_controller_atomics PRIVATE cxx_std_20)
target_compile_definitions(csx_controller_atomics PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(csx_controller_atomics PRIVATE /W4 /permissive- /EHsc)
endif()
